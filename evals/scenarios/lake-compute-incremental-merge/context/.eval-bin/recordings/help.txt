Find the models in this project that can run on Lake Compute

Usage: dbt init lakecompute [OPTIONS]

Options:
      --json
          Print the report as JSON (with --quiet, stdout is the report alone)

      --summary
          With --json, report counts, grouped blockers and the plan, not each model

      --all
          List every model, not just a summary

      --no-rewrite
          Don't rewrite SQL that Lake Compute rejects

      --switch
          Switch the models that can move: show the diff, then edit config and SQL

      --snowflake
          Compile on Snowflake (as dbt compile does), then size and price models from query history, EXPLAIN and table sizes (read-only queries)

      --history-days <HISTORY_DAYS>
          With --snowflake, the days of query history to read
          
          [default: 30]

      --dry-run
          Write nothing with --switch; print queries without running them with --snowflake

  -y, --yes
          With --switch or --snowflake, go ahead without asking

      --convert-views
          With --switch, also convert views that block models from moving

      --baseline
          Before switching, fingerprint each model's Snowflake table for --validate

      --validate
          After switched models build on Lake Compute, compare their tables with --baseline's

      --include-unverified
          With --switch, also switch models whose SQL couldn't be checked

      --run-results <RUN_RESULTS>
          Read build times from this run_results.json (of a `dbt run` or `dbt build`) [default: <target-path>/run_results.json]

      --warehouse-size <WAREHOUSE_SIZE>
          Warehouse size to price build times at
          
          [default: xsmall]
          [possible values: xsmall, small, medium, large, xlarge, 2xlarge, 3xlarge, 4xlarge, 5xlarge, 6xlarge]

      --maximum-seed-size-mib <MAXIMUM_SEED_SIZE_MIB>
          Maximum size (MiB) for seed files whose contents are hashed. Larger seeds fall back to a path-based checksum. Set to 0 to disable the limit (always hash contents). Defaults to 1 MiB when unset
          
          [env: DBT_MAXIMUM_SEED_SIZE_MIB=]

      --no-manage-state
          Disable service-backed dbt State

  -h, --help
          Print help (see a summary with '-h')

          Suppress all non-error logging to stdout. Does not affect {{ print() }} macro calls
          
          [env: DBT_QUIET=]

  -d, --debug
          Display debug logging during dbt execution. Useful for debugging and making bug reports
          
          [env: DBT_DEBUG=]

          [env: DBT_LOG_PATH=]

      --otel-file-name <OTEL_FILE_NAME>
          Set 'otel-file-name' for the current run, overriding 'DBT_OTEL_FILE_NAME'. If set, OTEL telemetry will be written to `$log_path/otel-file-name`
          
          [env: DBT_OTEL_FILE_NAME=]

      --otel-parquet-file-name <OTEL_PARQUET_FILE_NAME>
          Set 'otel-parquet-file-name' for the current run, overriding 'DBT_OTEL_PARQUET_FILE_NAME'. If set, OTEL telemetry will be written to `$target_path/metadata/otel-parquet-file-name` in Parquet format

      --log-format <LOG_FORMAT>
          Set logging format; use --log-format-file to override
          
          [env: DBT_LOG_FORMAT=]
          [default: default]
          [possible values: text, json, default, otel]

      --log-format-file <LOG_FORMAT_FILE>
          Set log file format, overriding the default and --log-format setting
          
          [env: DBT_LOG_FORMAT_FILE=]
          [possible values: text, json, default, otel]

      --log-level <LOG_LEVEL>
          Set minimum severity for console/log file; use --log-level-file to set log file severity separately
          
          [env: DBT_LOG_LEVEL=]
          [possible values: off, error, warn, info, debug, trace]

      --log-level-file <LOG_LEVEL_FILE>
          Set minimum log file severity, overriding the default and --log-level setting
          
          [env: DBT_LOG_LEVEL_FILE=]
          [possible values: off, error, warn, info, debug, trace]

      --invocation-id <INVOCATION_ID>
          Override the invocation ID with a UUID string or integer
          
          [env: DBT_INVOCATION_ID=]

      --parent-span-id <PARENT_SPAN_ID>
          Set the parent span ID for trace correlation (16-character hex or u64)
          
          [env: DBT_PARENT_SPAN_ID=]

Execution:
      --threads <THREADS>
          The number of threads to use [Run with --threads 0 to use max_cpu [default: max_cpu]]

      --compute <COMPUTE>
          Execution backend to use

          Possible values:
          - remote:  Execute on the remote warehouse (Snowflake, BigQuery, etc.)
          - inline:  Run computations in-process
          - sidecar: Run computations in a separate, ephemeral worker process
          - service: Run via the remote compute service (persistent workers/cluster)
          
          [env: DBT_COMPUTE=]
          [default: remote]

      --defer
          [env: DBT_DEFER=]

      --state <STATE>
          Unless overridden, use this state directory for both state comparison and deferral
          
          [env: DBT_STATE=]

      --dirty
          Select only nodes whose source files have changed or been added since the last --partial-parse run, plus all their downstream dependents. Implies --partial-parse. Use with --partial-load for full speed: --partial-load --dirty
          
          [env: DBT_DIRTY=]

      --no-version-check
          Disable online version check for dbt updates
          
          [env: DBT_DISABLE_VERSION_CHECK=]

      --store-failures
          Store test failures in the database (equivalent to dbt-core `--store-failures`)
          
          [env: DBT_STORE_FAILURES=]

      --manage-state
          Enable service-backed dbt State without legacy task-cache coordination
